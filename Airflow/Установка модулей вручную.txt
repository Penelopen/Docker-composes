# 1. Ставим в шедулер (hdfs, psycopg2-binary, apache-airflow-providers-apache-kafka, confluent-kafka)
docker exec -it airflow_airflow-scheduler_1 /home/airflow/.local/bin/pip install --no-deps hdfs==2.7.3 psycopg2-binary==2.9.9 apache-airflow-providers-apache-kafka==1.14.0 confluent-kafka

# 2. Ставим в воркер
docker exec -it airflow_airflow-worker_1 /home/airflow/.local/bin/pip install --no-deps hdfs==2.7.3 psycopg2-binary==2.9.9 apache-airflow-providers-apache-kafka==1.14.0 confluent-kafka

# 3. Ставим в веб-сервер
docker exec -it airflow_airflow-webserver_1 /home/airflow/.local/bin/pip install --no-deps hdfs==2.7.3 psycopg2-binary==2.9.9 apache-airflow-providers-apache-kafka==1.14.0 confluent-kafka

# Мягко перезапускаем процессы Airflow
docker-compose restart airflow-webserver airflow-scheduler

# Сбрасываем старый кэш ошибок в СУБД метастора
docker exec -it airflow_airflow-scheduler_1 airflow dags reserialize
