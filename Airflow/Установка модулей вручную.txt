# Ставим в шедулер, воркер и  веб-сервер модули hdfs, psycopg2-binary, apache-airflow-providers-apache-kafka, confluent-kafka

docker exec -it airflow-airflow-scheduler-1 /home/airflow/.local/bin/pip install --no-deps hdfs==2.7.3 psycopg2-binary==2.9.9 apache-airflow-providers-apache-kafka==1.4.1 confluent-kafka

docker exec -it airflow-airflow-worker-1 /home/airflow/.local/bin/pip install --no-deps hdfs==2.7.3 psycopg2-binary==2.9.9 apache-airflow-providers-apache-kafka==1.4.1 confluent-kafka

docker exec -it airflow-airflow-webserver-1 /home/airflow/.local/bin/pip install --no-deps hdfs==2.7.3 psycopg2-binary==2.9.9 apache-airflow-providers-apache-kafka==1.4.1 confluent-kafka

# Мягко перезапускаем процессы Airflow
docker-compose restart airflow-webserver airflow-scheduler

# Сбрасываем старый кэш ошибок в СУБД метастора
docker exec -it airflow-airflow-scheduler-1 airflow dags reserialize
